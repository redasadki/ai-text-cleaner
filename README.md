# 1. Go to your project folder
cd ~/github/ai-text-cleaner

# 2. Paste the clipboard content into a new file named README.md
pbpaste > README.md

# 3. Stage the new file for the next commit
git add README.md

# 4. Commit the file with a message
git commit -m "Added project documentation"

# 5. Push the changes to GitHub
git push
